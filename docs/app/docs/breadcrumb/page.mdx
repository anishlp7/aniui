## Usage

```tsx title="app/index.tsx"
import {
  Breadcrumb,
  BreadcrumbItem,
  BreadcrumbLink,
  BreadcrumbPage,
  BreadcrumbSeparator,
} from "@/components/ui/breadcrumb";

export function MyScreen() {
  return (
    <Breadcrumb>
      <BreadcrumbItem>
        <BreadcrumbLink onPress={() => {}}>Home</BreadcrumbLink>
      </BreadcrumbItem>
      <BreadcrumbSeparator />
      <BreadcrumbItem>
        <BreadcrumbPage>Breadcrumb</BreadcrumbPage>
      </BreadcrumbItem>
    </Breadcrumb>
  );
}
```

## Accessibility

- `BreadcrumbLink` uses `accessibilityRole="link"`.
- `BreadcrumbPage` uses `accessibilityRole="header"` to mark the current page.
