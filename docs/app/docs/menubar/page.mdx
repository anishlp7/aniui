## Usage

```tsx title="app/index.tsx"
import { Menubar, MenubarMenu, MenubarItem } from "@/components/ui/menubar";

export function MyScreen() {
  return (
    <Menubar>
      <MenubarMenu trigger="File">
        <MenubarItem onPress={() => {}}>New File</MenubarItem>
        <MenubarItem onPress={() => {}}>Save</MenubarItem>
      </MenubarMenu>
      <MenubarMenu trigger="Edit">
        <MenubarItem onPress={() => {}}>Undo</MenubarItem>
      </MenubarMenu>
    </Menubar>
  );
}
```

## Accessibility

- `Menubar` uses `accessibilityRole="menubar"`.
- Triggers and items use `accessibilityRole="menuitem"`; the open panel uses `accessibilityRole="menu"`.
